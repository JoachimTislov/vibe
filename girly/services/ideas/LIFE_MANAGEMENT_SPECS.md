#### Database Schema

```sql
-- Period tracking table
CREATE TABLE health_periods (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES girly_users.users(id) ON DELETE CASCADE,
    
    -- Period details
    start_date DATE NOT NULL,
    end_date DATE,
    duration_days INTEGER, -- Calculated
    
    -- Flow
    flow_level VARCHAR(20), -- light, medium, heavy, spotting
    
    -- Symptoms
    symptoms JSONB DEFAULT '{}', -- {"cramps": "severe", "mood_swings": "moderate"}
    symptom_intensity INTEGER DEFAULT 5 CHECK (symptom_intensity BETWEEN 1 AND 10),
    
    -- Notes
    notes TEXT,
    
    -- Metadata
    tags VARCHAR(50)[] DEFAULT '{}',
    
    -- Status
    is_predicted BOOLEAN DEFAULT FALSE,
    prediction_accuracy INTEGER, -- 0-100
    
    -- Timestamps
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW()
);

-- Cycle analytics cache
CREATE TABLE health_cycle_analytics (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES girly_users.users(id) ON DELETE CASCADE,
    
    -- Time period
    period VARCHAR(20) NOT NULL, -- current, last, all_time
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    
    -- Cycle statistics
    average_cycle_length_days DOUBLE PRECISION,
    cycle_regularity DOUBLE PRECISION, -- 0-1, 1 = very regular
    next_period_prediction DATE,
    prediction_confidence INTEGER DEFAULT 0 CHECK (prediction_confidence BETWEEN 0 AND 100),
    
    -- Ovulation
    ovulation_day INTEGER, -- Day of cycle
    ovulation_prediction_date DATE,
    
    -- Fertility window
    fertility_window_start DATE,
    fertility_window_end DATE,
    
    -- Symptoms
    common_symptoms JSONB DEFAULT '{}', -- {"cramps": 85, "bloating": 75}
    symptom_patterns JSONB DEFAULT '{}', -- Patterns over time
    
    -- Timestamps
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
    
    UNIQUE(user_id, period, start_date)
);

-- Symptoms table
CREATE TABLE health_symptoms (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES girly_users.users(id) ON DELETE CASCADE,
    
    -- Symptom details
    date DATE NOT NULL,
    time TIME WITH TIME ZONE,
    symptom_type VARCHAR(50) NOT NULL, -- physical or emotional
    symptom_name VARCHAR(100) NOT NULL,
    
    -- Severity
    severity VARCHAR(20) NOT NULL CHECK (severity IN ('mild', 'moderate', 'severe')),
    intensity INTEGER NOT NULL DEFAULT 5 CHECK (intensity BETWEEN 1 AND 10),
    
    -- Context
    related_to_cycle BOOLEAN DEFAULT FALSE,
    notes TEXT,
    
    -- Timestamps
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW()
);

-- Medications table
CREATE TABLE health_medications (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES girly_users.users(id) ON DELETE CASCADE,
    
    -- Medication details
    name VARCHAR(255) NOT NULL,
    type VARCHAR(50) NOT NULL CHECK (type IN ('prescription', 'over_the_counter', 'supplement', 'other')),
    purpose TEXT,
    
    -- Dosage
    dosage VARCHAR(100),
    frequency VARCHAR(100), -- e.g., "Once daily", "Twice a day"
    
    -- Schedule
    schedule JSONB NOT NULL, -- Array of {time, days_of_week}
    
    -- Reminders
    reminders_enabled BOOLEAN DEFAULT TRUE,
    reminder_minutes_before INTEGER DEFAULT 30,
    
    -- Status
    is_active BOOLEAN DEFAULT TRUE,
    start_date DATE,
    end_date DATE,
    
    -- Notes
    notes TEXT,
    
    -- Timestamps
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW()
);

-- Medication reminders table
CREATE TABLE health_medication_reminders (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    medication_id UUID NOT NULL REFERENCES health_medications(id) ON DELETE CASCADE,
    user_id UUID NOT NULL REFERENCES girly_users.users(id) ON DELETE CASCADE,
    
    -- Reminder details
    reminder_time TIME WITH TIME ZONE NOT NULL,
    reminder_date DATE NOT NULL,
    
    -- Status
    status VARCHAR(20) NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'sent', 'acknowledged', 'skipped')),
    
    -- Timestamps
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
    sent_at TIMESTAMP WITH TIME ZONE,
    acknowledged_at TIMESTAMP WITH TIME ZONE
);

-- Sleep tracking table
CREATE TABLE health_sleep (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES girly_users.users(id) ON DELETE CASCADE,
    
    -- Sleep details
    date DATE NOT NULL,
    bedtime TIME WITH TIME ZONE NOT NULL,
    wakeup_time TIME WITH TIME ZONE NOT NULL,
    
    -- Duration
    total_minutes INTEGER NOT NULL, -- Calculated
    
    -- Quality
    quality_rating INTEGER CHECK (quality_rating BETWEEN 1 AND 10),
    interruptions INTEGER DEFAULT 0,
    time_to_fall_asleep_minutes INTEGER,
    
    -- Notes
    notes TEXT,
    
    -- Timestamps
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
    
    UNIQUE(user_id, date)
);

-- Indexes
CREATE INDEX idx_health_periods_user ON health_periods(user_id);
CREATE INDEX idx_health_periods_start_date ON health_periods(start_date);
CREATE INDEX idx_health_symptoms_user ON health_symptoms(user_id);
CREATE INDEX idx_health_symptoms_date ON health_symptoms(date);
CREATE INDEX idx_health_medications_user ON health_medications(user_id);
CREATE INDEX idx_health_medication_reminders_user ON health_medication_reminders(user_id);
CREATE INDEX idx_health_medication_reminders_status ON health_medication_reminders(status);
CREATE INDEX idx_health_sleep_user ON health_sleep(user_id);
CREATE INDEX idx_health_sleep_date ON health_sleep(date);
```
